        print("Starting Code Generation...")

        manifest = load_yaml(DSL_DIR / "manifest.yaml")

        # Passo 1: construir entity_domain antes de gerar nada

        for domain, entities in entities_by_domain.items():

        if OPERATIONS_DIR.exists():
